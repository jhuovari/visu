Työmarkkinat

Työvoimatutkimuksen tärkeimmät tunnusluvut kuukausitasolla ja palkansaajien määrä vuositasolla. Alkuperäinen sarja kertoo, mitä kuukaudessa mitattiin, ja trendisarja sen, mihin suuntaan työmarkkinat ovat menossa. Ne ovat samassa kuviossa samalla värillä eri paksuisin viivoin, koska kyse on samasta ilmiöstä eri tavalla siloitettuna.

Tasot ovat suoraan tilastosta. Tilasto ei julkaise vuosimuutoksia, joten ne lasketaan tällä sivulla alkuperäisestä sarjasta 12 kuukauden takaiseen verrattuna — määristä henkinä ja asteista prosenttiyksikköinä. Saman sarjan taso ja muutos ovat kuvion välilehdillä, ja kuvion alta saa ruotsin- ja englanninkielisen version ladattavana kuvana.

library(visu)

tyottomyys <- pxwebtools::pxw_get_data(
  url = "https://pxdata.stat.fi/PxWeb/api/v1/fi/StatFin/tyti/135z.px/",
  query = list(
    "timeperiod_m" = "*",
    "contentscode" = c("tyti-Tyottomyysaste", "tyottaste_trendi")
  )
)

tekstit <- list(
  fi = list(
    otsikko = "Työttömyysaste, 15-74-vuotiaat, %",
    lahde = "Lähde: Tilastokeskus, työvoimatutkimus",
    sarjat = c("tyti-Tyottomyysaste" = "Alkuperäinen", "tyottaste_trendi" = "Trendi")
  ),
  sv = list(
    otsikko = "Relativt arbetslöshetstal, 15-74-åringar, %",
    lahde = "Källa: Statistikcentralen, arbetskraftsundersökningen",
    sarjat = c("tyti-Tyottomyysaste" = "Ursprunglig", "tyottaste_trendi" = "Trend")
  ),
  en = list(
    otsikko = "Unemployment rate, persons aged 15-74, %",
    lahde = "Source: Statistics Finland, Labour Force Survey",
    sarjat = c("tyti-Tyottomyysaste" = "Original", "tyottaste_trendi" = "Trend")
  )
)

kuvio <- function(kieli) {
  t <- tekstit[[kieli]]
  d <- tyottomyys
  d$sarja <- factor(d$contentscode, levels = names(t$sarjat), labels = t$sarjat)
  # Sama ilmio eri tavalla siloitettuna: sama vari, trendi paksummalla.
  visu_plot(d, linewidth = "sarja", title = t$otsikko, caption = t$lahde, lang = kieli)
}

kuvio("fi") |>
  visu_interactive(caption = tekstit$fi$lahde)

tyottomyys_muutos <- pxwebtools::pxw_get_data(

    otsikko = "Change in the unemployment rate from a year earlier, percentage points",
