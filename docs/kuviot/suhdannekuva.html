Suhdannekuva

Kaksi kuukausittain julkaistavaa mittaria, jotka kertovat talouden käänteestä ennen neljännesvuositilinpitoa. Tuotannon suhdannekuvaaja on kansantalouden tuotannon kuukausikuva, ja kuluttajien luottamusindikaattori kertoo miten kotitaloudet odottavat talouden kehittyvän.

Alkuperäinen, kausitasoitettu ja trendi ovat samassa kuviossa samalla värillä eri paksuisin viivoin: kyse on samasta ilmiöstä eri tavalla siloitettuna.

library(visu)

tuotanto <- pxwebtools::pxw_get_data(
  url = "https://pxdata.stat.fi/PxWeb/api/v1/fi/StatFin/ktkk/132f.px/",
  query = list(
    "timeperiod_m" = "*",
    "toimiala_79_20180101" = "SSS",
    "contentscode" = c("ind_tasm_2015", "ind_kausit_2015", "ind_trendi_2015")
  )
)

tekstit <- list(
  fi = list(
    otsikko = "Tuotannon suhdannekuvaaja, 2015 = 100",
    lahde = "Lähde: Tilastokeskus, tuotannon suhdannekuvaaja",
    sarjat = c(ind_tasm_2015 = "Alkuperäinen",
               ind_kausit_2015 = "Kausitasoitettu",
               ind_trendi_2015 = "Trendi")
  ),
  sv = list(
    otsikko = "Produktionens konjunkturindikator, 2015 = 100",
    lahde = "Källa: Statistikcentralen, produktionens konjunkturindikator",
    sarjat = c(ind_tasm_2015 = "Ursprunglig",
               ind_kausit_2015 = "Säsongrensad",
               ind_trendi_2015 = "Trend")
  ),
  en = list(
    otsikko = "Monthly indicator of total output, 2015 = 100",
    lahde = "Source: Statistics Finland, monthly indicator of total output",
    sarjat = c(ind_tasm_2015 = "Original",
               ind_kausit_2015 = "Seasonally adjusted",
               ind_trendi_2015 = "Trend")
  )
)

kuvio <- function(kieli) {
  t <- tekstit[[kieli]]
  d <- tuotanto
  # Karkeimmasta siloitetuimpaan, jolloin viiva paksunee jarjestyksessa.
  d$sarja <- factor(d$contentscode, levels = names(t$sarjat), labels = t$sarjat)
  visu_plot(d, linewidth = "sarja", title = t$otsikko, caption = t$lahde, lang = kieli)
}

kuvio("fi") |>
  visu_interactive(caption = tekstit$fi$lahde)
