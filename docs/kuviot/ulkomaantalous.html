Ulkomaantalous

Tiedot päivitetty 17.9.2026 klo 12:20.

Suomen talouden suhde ulkomaihin. Vaihtotase kertoo, ansaitseeko Suomi ulkomailta enemmän kuin maksaa sinne, ja tavaroiden ja palveluiden ulkomaankauppa erittelee, mistä virta syntyy.

Vaihtotase on kuukausisarjana hyvin heiluva, joten oletusnäkymä on 12 kuukauden liukuva summa. Yksittäiset kuukaudet ovat omalla välilehdellään.

Vaihtotase

  • 12 kuukauden summa
  • Kuukausittain

vaihtotase <- visu_get_data(
  url = "https://pxdata.stat.fi/PxWeb/api/v1/fi/StatFin/mata/114s.px/",
  query = list(

    "contentscode" = "NB12"
  )
)

  ),
  sv = list(

    selite = "Ett positivt tal är överskott",
    lahde = "Källa: Statistikcentralen, betalningsbalans och utlandsställning"

kuvio <- function(kieli) {
  t <- tekstit[[kieli]]
  visu_plot(vaihtotase, title = t$otsikko, subtitle = t$selite, caption = t$lahde,
            lang = kieli, zeroline = TRUE)
}

kuvio("fi") |>
  visu_interactive(subtitle = tekstit$fi$selite, caption = tekstit$fi$lahde)

library(visu)

kauppa <- visu_get_data(
  url = "https://pxdata.stat.fi/PxWeb/api/v1/fi/StatFin/tpulk/12gq.px/",
  query = list(
    "timeperiod_q" = "*",
    "valtio_27_20180101" = "ULK",
    "taloustoimi_1_20180101" = "GS",
    "contentscode" = c("tpulk_C", "tpulk_D")
  )
)

tekstit_kauppa <- list(
  fi = list(
    otsikko = "Tavaroiden ja palveluiden vienti ja tuonti, miljoonaa euroa",
    selite = "Ulkomaat yhteensä, neljännesvuosittain",
    lahde = "Lähde: Tilastokeskus, tavaroiden ja palveluiden ulkomaankauppa",
    sarjat = c(tpulk_C = "Vienti", tpulk_D = "Tuonti")
  ),
  sv = list(
    otsikko = "Export och import av varor och tjänster, miljoner euro",
    selite = "Utlandet totalt, kvartalsvis",
    lahde = "Källa: Statistikcentralen, utrikeshandel med varor och tjänster",
    sarjat = c(tpulk_C = "Export", tpulk_D = "Import")
  ),
  en = list(
    otsikko = "Exports and imports of goods and services, EUR million",
    selite = "Total foreign countries, by quarter",
    lahde = "Source: Statistics Finland, international trade in goods and services",
    sarjat = c(tpulk_C = "Exports", tpulk_D = "Imports")
  )
)

kuvio_kauppa <- function(kieli) {
  t <- tekstit_kauppa[[kieli]]
  d <- kauppa
  d$virta <- factor(d$contentscode, levels = names(t$sarjat), labels = t$sarjat)
  visu_plot(d, colour = "virta", title = t$otsikko,
            subtitle = t$selite, caption = t$lahde, lang = kieli)
}

kuvio_kauppa("fi") |>
  visu_interactive(subtitle = tekstit_kauppa$fi$selite,
                   caption = tekstit_kauppa$fi$lahde)
