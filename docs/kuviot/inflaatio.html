Inflaatio

Kolme mittaria samasta ilmiöstä. Kuluttajahintaindeksi (KHI) on kansallinen inflaatiomittari, yhdenmukaistettu kuluttajahintaindeksi (YKHI) EU-maiden vertailukelpoinen indeksi, ja YKHI:n ennakko on kuukauden vaihteessa julkaistava pikaestimaatti noin kaksi viikkoa ennen varsinaista julkistusta. YKHI kiintein veroin (YKHI-KIVE) kertoo, paljonko inflaatiota jäisi jäljelle ilman välillisten verojen muutoksia.

Saman sarjan taso ja muutos ovat kuvion välilehdillä. Kuvion alta saa ruotsin- ja englanninkielisen version ladattavana kuvana, ja kuvion tiedot kertovat mistä taulusta luvut ovat.

muutos <- rbind(
  data.frame(time = khi$time, sarja = "khi", values = khi$values),
  data.frame(time = ykhi$time, sarja = "ykhi", values = ykhi$values),
  data.frame(time = ennakko$time, sarja = "ennakko", values = ennakko$values)
)

# Ennakkotaulu kattaa vain kuluvan vuoden, joten koko historia hukuttaisi sen.
alku <- seq(max(muutos$time), by = "-8 years", length.out = 2)[2]
muutos <- muutos[muutos$time >= alku, ]

# Kaannokset elavat kuvion koodin vieressa: sama funktio piirtaa seka sivun
# suomenkielisen kuvion etta ladattavat kieliversiot.
tekstit <- list(
  fi = list(
    otsikko = "Kuluttajahintojen vuosimuutos, %",
    lahde = "Lähde: Tilastokeskus, kuluttajahintaindeksi",
    sarjat = c(khi = "Kuluttajahintaindeksi (KHI)",
               ykhi = "Yhdenmukaistettu (YKHI)",
               ennakko = "YKHI, ennakko")
  ),
  sv = list(
    otsikko = "Konsumentprisernas årsförändring, %",
    lahde = "Källa: Statistikcentralen, konsumentprisindex",
    sarjat = c(khi = "Konsumentprisindex (KPI)",
               ykhi = "Harmoniserat (HKPI)",
               ennakko = "HKPI, preliminär")
  ),
  en = list(
    otsikko = "Annual change in consumer prices, %",
    lahde = "Source: Statistics Finland, Consumer Price Index",
    sarjat = c(khi = "Consumer Price Index (CPI)",
               ykhi = "Harmonised (HICP)",
               ennakko = "HICP, preliminary")
  )
)

kuvio <- function(kieli) {
  t <- tekstit[[kieli]]
  d <- muutos
  d$mittari <- factor(d$sarja, levels = names(t$sarjat), labels = t$sarjat)
  visu_plot(d, colour = "mittari", title = t$otsikko, caption = t$lahde, lang = kieli,
            zeroline = TRUE)
}

kuvio("fi") |>
  visu_interactive(caption = tekstit$fi$lahde)
