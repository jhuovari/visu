Neljännesvuositilinpito

Kansantalouden neljännesvuositilinpidon keskeiset erät. Tasot ovat kausitasoitettuja ja työpäiväkorjattuja volyymisarjoja viitevuoden 2015 hinnoin, ja muutokset saman sarjan volyymin muutoksia vuodentakaisesta. Kausitasoitettu sarja kertoo suhdannekäänteen aikaisimmin, trendisarja luotettavimmin.

Saman sarjan taso ja muutos ovat kuvion välilehdillä, ja kuvion alta saa ruotsin- ja englanninkielisen version ladattavana kuvana.

library(visu)

bkt <- pxwebtools::pxw_get_data(
  url = "https://pxdata.stat.fi/PxWeb/api/v1/fi/StatFin/ntp/132h.px/",
  query = list(
    "timeperiod_q" = "*",
    "taloustoimi_1_20180101" = "B1GMH",
    "contentscode" = c("kausitvv2015", "trendivv2015")
  )
)

# Koko historiassa sarjat kulkevat paallekkain, joten taso rajataan viimeiseen
# kymmeneen vuoteen. Pitka kuva on vuosimuutoksessa ja kysynnan paaerissa.
alku <- seq(max(bkt$time), by = "-10 years", length.out = 2)[2]
bkt <- bkt[bkt$time >= alku, ]

tekstit <- list(
  fi = list(
    otsikko = "Bruttokansantuotteen volyymi, miljoonaa euroa",
    selite = "Viitevuosi 2015",
    lahde = "Lähde: Tilastokeskus, neljännesvuositilinpito",
    sarjat = c(kausitvv2015 = "Kausitasoitettu", trendivv2015 = "Trendi")
  ),
  sv = list(
    otsikko = "Bruttonationalproduktens volym, miljoner euro",
    selite = "Referensår 2015",
    lahde = "Källa: Statistikcentralen, kvartalsräkenskaper",
    sarjat = c(kausitvv2015 = "Säsongrensad", trendivv2015 = "Trend")
  ),
  en = list(
    otsikko = "Volume of gross domestic product, EUR million",
    selite = "Reference year 2015",
    lahde = "Source: Statistics Finland, quarterly national accounts",
    sarjat = c(kausitvv2015 = "Seasonally adjusted", trendivv2015 = "Trend")
  )
)

kuvio <- function(kieli) {
  t <- tekstit[[kieli]]
  d <- bkt
  d$sarja <- factor(d$contentscode, levels = names(t$sarjat), labels = t$sarjat)
  visu_plot(d, colour = "sarja", title = t$otsikko,
            subtitle = t$selite, caption = t$lahde)
}

kuvio("fi") |>
  visu_interactive(subtitle = tekstit$fi$selite, caption = tekstit$fi$lahde)
