Kotitaloudet

Kotitalouksien taloudellinen asema kahdesta suunnasta: paljonko tuloista jää säästöön ja paljonko velkaa on suhteessa tuloihin. Säästämisaste on sektoritilien tunnusluku ja velkaantumisaste rahoitustilinpidon.

Säästämisaste

  • Kotitaloudet

library(visu)

saastaminen <- pxwebtools::pxw_get_data(
  url = "https://pxdata.stat.fi/PxWeb/api/v1/fi/StatFin/ntp/15a8.px/",
  query = list(
    "timeperiod_q" = "*",
    "sektoriluokitus_7_20230101" = "S14",
    "contentscode" = c("ntp-tasm_b8n", "ntp-kausit_b8n")
  )
)

tekstit <- list(
  fi = list(
    otsikko = "Kotitalouksien säästämisaste, %",
    selite = "Säästö suhteessa käytettävissä olevaan tuloon",
    lahde = "Lähde: Tilastokeskus, kansantalouden sektoritilit",
    sarjat = c("ntp-tasm_b8n" = "Alkuperäinen", "ntp-kausit_b8n" = "Kausitasoitettu")
  ),
  sv = list(
    otsikko = "Hushållens sparkvot, %",
    selite = "Sparande i förhållande till disponibel inkomst",
    lahde = "Källa: Statistikcentralen, sektorräkenskaper",
    sarjat = c("ntp-tasm_b8n" = "Ursprunglig", "ntp-kausit_b8n" = "Säsongrensad")
  ),
  en = list(
    otsikko = "Household saving rate, %",
    selite = "Saving relative to disposable income",
    lahde = "Source: Statistics Finland, sector accounts",
    sarjat = c("ntp-tasm_b8n" = "Original", "ntp-kausit_b8n" = "Seasonally adjusted")
  )
)

kuvio <- function(kieli) {
  t <- tekstit[[kieli]]
  d <- saastaminen
  # Sama ilmio eri tavalla siloitettuna: sama vari, kausitasoitettu paksummalla.
  d$sarja <- factor(d$contentscode, levels = names(t$sarjat), labels = t$sarjat)
  visu_plot(d, linewidth = "sarja", title = t$otsikko,
            subtitle = t$selite, caption = t$lahde, lang = kieli)
}

kuvio("fi") |>
  visu_interactive(subtitle = tekstit$fi$selite, caption = tekstit$fi$lahde)
